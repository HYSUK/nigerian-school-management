    <script>
        async function handleLogin(event) {
            event.preventDefault();
            const loginBtn = document.querySelector('.btn-login');
            const originalText = loginBtn.innerHTML;
            
            loginBtn.disabled = true;
            loginBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Verifying...';

            const user = document.getElementById('username').value.trim();
            const pass = document.getElementById('password').value.trim();

            try {
                const response = await fetch('/api/v1/auth/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ username: user, password: pass })
                });

                const result = await response.json();

                if (!response.ok) throw new Error(result.detail || "Authentication rejected.");

                // Save application session parameters permanently in browser cache
                localStorage.setItem('active_auth_app_id', result.application_id);
                localStorage.setItem('active_auth_class', result.target_class);

                alert("🔓 Access Granted: " + result.message);

                // Direct routing switch based on profile completion rules
                if (result.is_profile_pending) {
                    // Redirect straight to wizard forms to complete missing demographics
                    window.location.href = "/register";
                } else {
                    // Send to standard student metrics console dashboard
                    window.location.href = "/dashboard";
                }

            } catch (error) {
                alert("❌ Authentication Failed: " + error.message);
            } finally {
                loginBtn.disabled = false;
                loginBtn.innerHTML = originalText;
            }
        }
    </script>
