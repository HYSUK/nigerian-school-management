    <script>
        async function checkStatus() {
  const token = document.getElementById("dashboardSearchField").value.trim();
  if (!token) return;

  try {
    const response = await fetch(`/api/v1/student/${token}`);
    if (!response.ok) throw new Error("Profile not found");

    const data = await response.json();

    // Greeting
    document.getElementById("studentName").innerText = data.first_name;
    document.getElementById("greeting").classList.remove("hidden");

    // Student Image
    if (data.passport_base64) {
      document.getElementById("studentImage").src = "data:image/png;base64," + data.passport_base64;
      document.getElementById("studentImageWrapper").classList.remove("hidden");
    }

    // Status Card
    document.getElementById("resName").innerText = data.first_name + " " + data.last_name;
    document.getElementById("resClass").innerText = data.target_class;

    const statusLabel = document.getElementById("resStatus");
    statusLabel.innerText = "PASSED SCREENING";
    statusLabel.className = "px-2 py-0.5 rounded text-xs font-bold bg-green-100 text-green-800";

    document.getElementById("resultCard").classList.remove("hidden");
    document.getElementById("printActionWrapper").classList.remove("hidden");
  } catch (err) {
    alert("Error: " + err.message);
  }
}
    </script>
