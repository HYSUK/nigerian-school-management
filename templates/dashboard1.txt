    <script>
        const API_BASE = "/api/v1";
        let cachedAppId = null;

        // Window Lifecycle initialization block hooks
        window.addEventListener('DOMContentLoaded', () => {
            cachedAppId = localStorage.getItem('portal_app_id');
            if(!cachedAppId) {
                window.location.href = "/home";
                return;
            }
            fetchFreshServerMetrics();
        });

        // Pull fresh tracking data directly from the server matching the current cache session identity
        async function fetchFreshServerMetrics() {
            try {
                const response = await fetch(`${API_BASE}/student/${cachedAppId}`);
                if (!response.ok) throw new Error("Synchronization fetch cluster rejected data pipeline request.");
                const res = await response.json();
                
                if(res.success) {
                    populateDashboardLayout(res);
                }
            } catch (err) {
                console.error("Sync Error:", err);
                // Fall back gracefully onto localized client storage variables if the network is flaky
                populateLayoutWithLocalFallback();
            }
        }

        function populateDashboardLayout(data) {
            const combinedName = `${data.first_name} ${data.last_name}`;
            
            // Map strings across UI positions
            document.getElementById('topBannerStudentName').innerText = combinedName;
            document.getElementById('sidebarStudentName').innerText = combinedName;
            document.getElementById('topBannerClass').innerText = data.target_class;
            
            document.getElementById('sidebarAppId').innerText = data.application_id;
            document.getElementById('cardAppId').innerText = data.application_id;
            document.getElementById('cardFirstName').innerText = data.first_name;
            document.getElementById('cardLastName').innerText = data.last_name;
            document.getElementById('cardClass').innerText = data.target_class;

            // Handle system rendering logic based on image path responses
            if (data.passport_url) {
                const navAvatar = document.getElementById('sidebarAvatar');
                const bannerAvatar = document.getElementById('topBannerAvatar');
                
                navAvatar.src = data.passport_url;
                bannerAvatar.src = data.passport_url;
                
                navAvatar.classList.remove('hidden');
                bannerAvatar.classList.remove('hidden');
                document.getElementById('sidebarPlaceholder').classList.add('hidden');
                document.getElementById('topBannerPlaceholder').classList.add('hidden');
            }

            // Sync Badge status changes dynamically
            const badge = document.getElementById('topBannerBadge');
            if (data.admission_status.includes("Approved")) {
                badge.innerText = "Approved & Synced";
                badge.className = "self-start sm:self-center px-4 py-1.5 rounded-full text-xs font-bold tracking-wide shadow-sm uppercase bg-emerald-100 text-emerald-800 border border-emerald-200";
            } else {
                badge.innerText = "Awaiting Screening";
                badge.className = "self-start sm:self-center px-4 py-1.5 rounded-full text-xs font-bold tracking-wide shadow-sm uppercase bg-amber-100 text-amber-800 border border-amber-200 animate-pulse";
            }
        }

        function populateLayoutWithLocalFallback() {
            const fName = localStorage.getItem('portal_first_name') || 'Student';
            const lName = localStorage.getItem('portal_last_name') || '';
            const cls = localStorage.getItem('portal_class') || 'Pending';
            
            const stubData = {
                application_id: cachedAppId,
                first_name: fName,
                last_name: lName,
                target_class: cls,
                passport_url: localStorage.getItem('portal_passport') || null,
                admission_status: "Local Cache View (Offline Mode)"
            };
            populateDashboardLayout(stubData);
        }

        // Dashboard Single Page Panel toggle switches
        function switchView(targetPanel) {
            const panels = ['gateway', 'letter'];
            panels.forEach(p => {
                const el = document.getElementById(`panel_${p}`);
                const menuBtn = document.getElementById(`menu_${p}`);
                
                if (p === targetPanel) {
                    el.classList.remove('hidden');
                    menuBtn.className = "w-full flex items-center space-x-3 px-4 py-3 rounded-xl font-semibold text-sm transition bg-slate-800 text-white";
                } else {
                    el.classList.add('hidden');
                    menuBtn.className = "w-full flex items-center space-x-3 px-4 py-3 rounded-xl font-semibold text-sm transition text-slate-400 hover:bg-slate-800/50 hover:text-slate-200";
                }
            });
        }

        // Native streaming handler: Download PDF letters via specific identification codes
        function downloadAdmissionLetter() {
            // Converts characters like "/" cleanly to match route security mapping signatures
            const encodedId = cachedAppId.replace(/\//g, "_");
            window.open(`${API_BASE}/admission-letter/${encodedId}`, '_blank');
        }

        // Flush credentials and session states safely on exit
        function handleLogout() {
            localStorage.clear();
            window.location.href = "/home";
        }
