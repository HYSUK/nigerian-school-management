    <script>
        let currentStep = 1;
        let globalPassportBase64 = null;

        // Dictionary containing sample State-to-LGA mapping data matching standard structures
        const NIGERIAN_GEO_MAP = {
            "Kano": ["Fagge", "Dala", "Gwale", "Nassarawa", "Tarauni", "Kano Municipal", "Ungogo", "Kumbotso"],
            "Lagos": ["Ikeja", "Alimosho", "Ojo", "Ikorodu", "Lagos Island", "Epe", "Badagry", "Surulere"],
            "FCT Abuja": ["Garki", "Wuse", "Asokoro", "Maitama", "Bwari", "Kuje", "Gwagwalada"],
            "Kaduna": ["Kaduna North", "Kaduna South", "Chikun", "Igabi", "Zaria", "Sabon Gari"]
        };

        // Window Lifecycle initialization wrapper
        window.addEventListener('DOMContentLoaded', () => {
            const appId = localStorage.getItem('portal_app_id');
            const cls = localStorage.getItem('portal_class');

            if(!appId) {
                window.location.href = "/home";
                return;
            }

            document.getElementById('headerAppId').innerText = appId;
            document.getElementById('headerClass').innerText = cls;

            // Trigger structural view changes for Senior Secondary candidates
            if(cls && cls.toUpperCase().startsWith("SS")) {
                document.getElementById('juniorHighHistoryContainer').classList.remove('hidden');
            }

            // Populate States array dropdown
            const stateDropdown = document.getElementById('stateOfOrigin');
            Object.keys(NIGERIAN_GEO_MAP).forEach(s => {
                let op = new Option(s, s);
                stateDropdown.add(op);
            });
        });

        function populateLGAs() {
            const state = document.getElementById('stateOfOrigin').value;
            const lgaDropdown = document.getElementById('lga');
            lgaDropdown.innerHTML = '<option value="" disabled selected>-- Select LGA --</option>';
            
            if(NIGERIAN_GEO_MAP[state]) {
                NIGERIAN_GEO_MAP[state].forEach(l => {
                    lgaDropdown.add(new Option(l, l));
                });
            }
        }

        // Client-side HTML5 Canvas Base64 Compression Engine
        function processPassportUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    
                    // Force downscaling to standard portrait proportions (e.g., 300x300 pixels)
                    canvas.width = 300;
                    canvas.height = 300;
                    ctx.drawImage(img, 0, 0, 300, 300);
                    
                    // Convert canvas structure to highly compressed raw base64 output string
                    globalPassportBase64 = canvas.toDataURL('image/jpeg', 0.75);
                    
                    // Render image asset preview
                    document.getElementById('passportPlaceholderIcon').classList.add('hidden');
                    const imgPreview = document.getElementById('passportImgPreview');
                    imgPreview.src = globalPassportBase64;
                    imgPreview.classList.remove('hidden');
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        }

        // Stepper Navigation Management Logic
        function navigateStep(direction) {
            // Run field validations before advancing forward
            if (direction === 1 && !validateCurrentStepFields()) {
                showWizardAlert("Please fill out all fields on this page before continuing.");
                return;
            }
            hideWizardAlert();

            // Toggle old panels off
            document.getElementById(`stepPanel${currentStep}`).classList.add('hidden');
            document.getElementById(`stepIndicator${currentStep}`).classList.remove('text-emerald-600');

            currentStep += direction;

            // Toggle new panels on
            document.getElementById(`stepPanel${currentStep}`).classList.remove('hidden');
            document.getElementById(`stepIndicator${currentStep}`).classList.add('text-emerald-600');

            // Button Controller evaluations
            document.getElementById('btnBack').style.visibility = (currentStep === 1) ? 'hidden' : 'visible';
            
            if (currentStep === 3) {
                document.getElementById('btnNext').classList.add('hidden');
                document.getElementById('btnSubmitForm').classList.remove('hidden');
            } else {
                document.getElementById('btnNext').classList.remove('hidden');
                document.getElementById('btnSubmitForm').classList.add('hidden');
            }
        }

        function validateCurrentStepFields() {
            const panel = document.getElementById(`stepPanel${currentStep}`);
            const inputs = panel.querySelectorAll('input[required], select[required], textarea[required]');
            for (let input of inputs) {
                if (!input.value.trim()) return false;
            }
            if (currentStep === 1 && !globalPassportBase64) {
                showWizardAlert("An official passport photo attachment is required.");
                return false;
            }
            return true;
        }

        function showWizardAlert(msg, mode='error') {
            const b = document.getElementById('wizardAlert');
            b.innerText = msg;
            b.className = `m-6 p-4 rounded-xl text-sm font-medium border ${mode==='error' ? 'bg-red-50 border-red-200 text-red-800':'bg-emerald-50 border-emerald-200 text-emerald-800'}`;
            b.classList.remove('hidden');
        }
        function hideWizardAlert() { document.getElementById('wizardAlert').add('hidden'); }

        // Endpoint Submission Pipeline: Handshake with /api/v1/register
        async function handleFormSubmission(e) {
            e.preventDefault();
            
            const btn = document.getElementById('btnSubmitForm');
            btn.disabled = true;
            btn.innerText = "Transmitting Demographics Profile...";

            const payload = {
                application_id: localStorage.getItem('portal_app_id'),
                first_name: document.getElementById('firstName').value,
                last_name: document.getElementById('lastName').value,
                gender: document.getElementById('gender').value,
                date_of_birth: document.getElementById('dob').value,
                state_of_origin: document.getElementById('stateOfOrigin').value,
                lga: document.getElementById('lga').value,
                home_address: document.getElementById('homeAddress').value,
                guardian_name: document.getElementById('guardianName').value,
                guardian_relationship: document.getElementById('guardianRelationship').value,
                guardian_phone: document.getElementById('guardianPhone').value,
                prev_primary_school: document.getElementById('prevPrimary').value || null,
                primary_from_year: document.getElementById('primaryFrom').value || null,
                primary_to_year: document.getElementById('primaryTo').value || null,
                prev_junior_sec_school: document.getElementById('prevJunior').value || null,
                junior_sec_from_year: document.getElementById('juniorFrom').value || null,
                junior_sec_to_year: document.getElementById('juniorTo').value || null,
                passport_base64: globalPassportBase64
            };

            try {
                const response = await fetch('/api/v1/register', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const data = await response.json();
                if(!response.ok) throw new Error(data.detail || "Profile registration pipeline error.");

                // Overwrite updated data metrics to client storage context caches
                localStorage.setItem('portal_first_name', payload.first_name);
                localStorage.setItem('portal_last_name', payload.last_name);

                showWizardAlert("Profile updated and synchronized successfully! Redirecting...", 'success');
                setTimeout(() => {
                    window.location.href = "/dashboard";
                }, 2000);

            } catch (err) {
                showWizardAlert(err.message, 'error');
                btn.disabled = false;
                btn.innerHTML = `<span>Submit Complete Registration</span><i class="fa-solid fa-cloud-arrow-up"></i>`;
            }
        }
    </script>
